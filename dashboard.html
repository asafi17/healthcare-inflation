Healthcare Services Price Index Summary

This section tracks headline healthcare services price indices from three major U.S. statistical programs. Each table reports the price index level, year-over-year (YoY) change, and month-over-month (MoM) change for the most recent four months and a trailing 12-month average. Because these series are released on different schedules, “Month \(t\)” reflects each source’s own most recent available data.

For a given price index \(P_t\), the measures are computed as:

\[ \text{YoY}_t = \left(\frac{P_t}{P_{t-12}} - 1\right) \times 100 \]

\[ \text{MoM}_t = \left(\frac{P_t}{P_{t-1}} - 1\right) \times 100 \]

The four headline series are:

Source Haver Code Description
PCE Price Index jcsdm@USNA Healthcare Services PCE Price Index (SA)
CPI UMS@CPIDATA CPI-U Medical Care Services (SA)
PPI Commodity P51@PPI PPI Healthcare Services, Commodity-Based (NSA)
PPI Industry RHC@PPIR PPI Selected Health Care Industries, Industry-Based (SA)
Code 
# --------------------------------------------------------------------------
# Helper: build a summary table for a single headline price index series
# --------------------------------------------------------------------------
build_summary_table <- function(series_code, database, label) {
  full_code <- paste0(series_code, "@", database)
  d <- import_haver(series = full_code, eop = FALSE)
  col <- setdiff(names(d), "date")
  names(d)[names(d) == col[1]] <- "pi"

  d <- d |>
    arrange(date) |>
    filter(!is.na(pi)) |>
    mutate(
      yoy = (pi / lag(pi, 12) - 1) * 100,
      mom = (pi / lag(pi, 1) - 1) * 100
    )

  # Most recent month for this series
  latest <- d |> filter(!is.na(yoy)) |> slice_tail(n = 1)
  latest_date <- latest$date

  # Get the 4 most recent months with complete data
  recent <- d |>
    filter(!is.na(yoy), date <= latest_date) |>
    slice_tail(n = 4)

  # 12-month average (t-1 to t-13, excluding the most recent month)
  trailing_12 <- d |>
    filter(!is.na(yoy), date <= latest_date) |>
    slice_tail(n = 13) |>
    slice_head(n = 12)

  avg_pi  <- mean(trailing_12$pi)
  avg_yoy <- mean(trailing_12$yoy)
  avg_mom <- mean(trailing_12$mom, na.rm = TRUE)

  # Build the display columns: t, t-1, t-2, t-3, 12-mo avg
  col_labels <- c(
    format(recent$date[4], "%b %Y"),
    format(recent$date[3], "%b %Y"),
    format(recent$date[2], "%b %Y"),
    format(recent$date[1], "%b %Y"),
    "12-Mo Avg"
  )

  tbl <- data.frame(
    Measure = c("Price Index", "YoY Change (%)", "MoM Change (%)"),
    t0 = c(
      round(recent$pi[4], 2),
      round(recent$yoy[4], 2),
      round(recent$mom[4], 2)
    ),
    t1 = c(
      round(recent$pi[3], 2),
      round(recent$yoy[3], 2),
      round(recent$mom[3], 2)
    ),
    t2 = c(
      round(recent$pi[2], 2),
      round(recent$yoy[2], 2),
      round(recent$mom[2], 2)
    ),
    t3 = c(
      round(recent$pi[1], 2),
      round(recent$yoy[1], 2),
      round(recent$mom[1], 2)
    ),
    avg = c(
      round(avg_pi, 2),
      round(avg_yoy, 2),
      round(avg_mom, 2)
    )
  )

  names(tbl) <- c("Measure", col_labels)

  list(table = tbl, label = paste0(label, " (", series_code, "@", database, ")"))
}

Healthcare Services Contribution to Core PCE Inflation

This section measures how much of Core PCE inflation is attributable to healthcare services. The approach follows Shria Holla’s code in healthcare_graphs.Rmd (section 5).

Core PCE excludes food and energy and can be decomposed into three mutually exclusive components:

Component Haver NE Code Haver PI Code Description
Goods ex Food & Energy cgxfem@USNA jcgxfem@USNA Core goods (excluding food and energy)
Services ex Energy & Healthcare csxhem@USNA jcsxhem@USNA Core services (excluding energy and healthcare)
Healthcare Services csdm@USNA jcsdm@USNA PCE healthcare services
Core PCE (aggregate) cxfem@USNA jcxfem@USNA Core PCE total

For each component \(i\), we first compute the YoY inflation rate from its price index \(P_{i,t}\):

\[ \pi_{i,t} = \left(\frac{P_{i,t}}{P_{i,t-12}} - 1\right) \times 100 \]

Next, we compute each component’s expenditure share of Core PCE using nominal expenditures (\(NE\)):

\[ w_{i,t} = \frac{NE_{i,t}}{NE_{core,t}} \]

Each component’s contribution to Core PCE inflation (in percentage points) is the product of its share and its inflation rate:

\[ C_{i,t} = w_{i,t} \times \pi_{i,t} \]

Estimated Core PCE inflation is the sum of all component contributions:

\[ \hat{\pi}_{core,t} = C_{goodxfe,t} + C_{servxeh,t} + C_{hcserv,t} \]

Finally, healthcare’s share of Core PCE inflation expresses healthcare’s contribution as a percentage of the total:

\[ S_{hcserv,t} = \frac{C_{hcserv,t}}{\hat{\pi}_{core,t}} \times 100 \]

Code 
# --------------------------------------------------------------------------
# Step 1: Pull nominal expenditures and price indices from Haver
# --------------------------------------------------------------------------
# Core PCE is decomposed into three components:
#   - goodxfe: Goods excluding Food & Energy
#   - servxeh: Services excluding Energy & Healthcare
#   - hcserv:  Healthcare Services
#
# For each we need nominal expenditures (NE) to compute shares,
# and price indices (PI) to compute inflation rates.

core_data <- import_haver(
  series = c(
    "cxfem@USNA",   # Core PCE NE
    "cgxfem@USNA",  # Goods ex F&E NE
    "csxhem@USNA",  # Services ex E&HC NE
    "csdm@USNA",    # HC Services NE
    "jcxfem@USNA",  # Core PCE PI
    "jcgxfem@USNA", # Goods ex F&E PI
    "jcsxhem@USNA", # Services ex E&HC PI
    "jcsdm@USNA"    # HC Services PI
  ),
  eop = FALSE
) |>
  rename(
    core_ne    = cxfem,
    goodxfe_ne = cgxfem,
    servxeh_ne = csxhem,
    hcserv_ne  = csdm,
    core_pi    = jcxfem,
    goodxfe_pi = jcgxfem,
    servxeh_pi = jcsxhem,
    hcserv_pi  = jcsdm
  ) |>
  arrange(date)

# --------------------------------------------------------------------------
# Step 2: Compute YoY inflation for each component
# --------------------------------------------------------------------------
# YoY inflation = ((PI_t / PI_{t-12}) - 1) * 100
# This gives the percentage change in prices over the past 12 months.

components <- c("core", "goodxfe", "servxeh", "hcserv")

for (comp in components) {
  pi_col   <- paste0(comp, "_pi")
  infl_col <- paste0(comp, "_infl")
  core_data <- core_data |>
    mutate(!!infl_col := ((get(pi_col) / lag(get(pi_col), 12)) - 1) * 100)
}

# --------------------------------------------------------------------------
# Step 3: Compute expenditure shares relative to Core PCE
# --------------------------------------------------------------------------
# Each component's share = its nominal expenditure / Core PCE nominal expenditure.
# These shares reflect how much of Core PCE spending goes to each category.

for (comp in c("goodxfe", "servxeh", "hcserv")) {
  ne_col    <- paste0(comp, "_ne")
  share_col <- paste0(comp, "_expshare")
  core_data <- core_data |>
    mutate(!!share_col := get(ne_col) / core_ne)
}

# --------------------------------------------------------------------------
# Step 4: Compute contributions (share * inflation)
# --------------------------------------------------------------------------
# A component's contribution to aggregate inflation = its expenditure share
# multiplied by its own inflation rate. This tells us how many percentage
# points of Core PCE inflation are driven by each component.

for (comp in c("goodxfe", "servxeh", "hcserv")) {
  infl_col    <- paste0(comp, "_infl")
  share_col   <- paste0(comp, "_expshare")
  contrib_col <- paste0(comp, "_contrib")
  core_data <- core_data |>
    mutate(!!contrib_col := get(share_col) * get(infl_col))
}

# --------------------------------------------------------------------------
# Step 5: Estimate total Core PCE inflation from contributions
# --------------------------------------------------------------------------
# Summing the three contributions gives an estimate of Core PCE inflation.
# This should closely match the directly computed core_infl.

core_data <- core_data |>
  mutate(core_infl_est = goodxfe_contrib + servxeh_contrib + hcserv_contrib)

# --------------------------------------------------------------------------
# Step 6: Healthcare's share of Core PCE inflation
# --------------------------------------------------------------------------
# What percentage of total Core PCE inflation is attributable to healthcare?
# hcserv_contrib_share = (hcserv_contrib / core_infl_est) * 100

core_data <- core_data |>
  mutate(hcserv_contrib_share = (hcserv_contrib / core_infl_est) * 100)

Medical Care Services Contribution to Core CPI Inflation

This section measures how much of Core CPI inflation is attributable to medical care services, mirroring the PCE contribution analysis above.

Core CPI (All Items Less Food & Energy) includes medical care services as a sub-component. We measure medical care services’ contribution using BLS relative importance weights, which reflect each component’s expenditure share of the total index.

Series Haver Code Description
Core CPI Price Index UIXFDG@CPIDATA CPI-U All Items Less Food & Energy (SA)
Medical Care Services Price Index UMS@CPIDATA CPI-U Medical Care Services (SA)
Medical Care Services RI Weight RPCUMS@CPIDATA Relative importance weight, Medical Care Services
Core CPI RI Weight RUIXFDGM@CPIDATA Relative importance weight, All Items Less Food & Energy

For each price index \(P_t\), we compute year-over-year inflation:

\[ \pi_t = \left(\frac{P_t}{P_{t-12}} - 1\right) \times 100 \]

Medical care services’ weight within Core CPI is obtained by rescaling the BLS relative importance weights (which are shares of All Items CPI) to shares of Core CPI:

\[ w_{ms,t} = \frac{RI^{ms}_t}{RI^{core}_t} \]

The contribution (in percentage points) and share of Core CPI inflation are:

\[ C_{ms,t} = w_{ms,t} \times \pi_{ms,t}, \qquad S_{ms,t} = \frac{C_{ms,t}}{\pi_{core,t}} \times 100 \]

Code 
# --------------------------------------------------------------------------
# Step 1: Pull price indices and relative importance weights from Haver
# --------------------------------------------------------------------------
cpi_core_data <- import_haver(
  series = c(
    "UIXFDG@CPIDATA",   # Core CPI PI (All Items Less Food & Energy, SA)
    "UMS@CPIDATA",       # Medical Care Services PI (SA)
    "RPCUMS@CPIDATA",    # RI weight: Medical Care Services
    "RUIXFDGM@CPIDATA"   # RI weight: All Items Less Food & Energy
  ),
  eop = FALSE
) |>
  rename(
    core_pi    = uixfdg   ,
    ms_pi      = ums,
    ms_ri      = rpcums,
    core_ri    = ruixfdgm
  ) |>
  arrange(date)

# --------------------------------------------------------------------------
# Step 2: Compute YoY inflation for Core CPI and Medical Care Services
# --------------------------------------------------------------------------
cpi_core_data <- cpi_core_data |>
  mutate(
    core_infl = ((core_pi / lag(core_pi, 12)) - 1) * 100,
    ms_infl   = ((ms_pi / lag(ms_pi, 12)) - 1) * 100
  )

# --------------------------------------------------------------------------
# Step 3: Rescale RI weights to get Medical Care Services' share of Core CPI
# --------------------------------------------------------------------------
# BLS relative importance weights are shares of All Items CPI.
# Dividing by the Core CPI weight gives the share within Core CPI.
cpi_core_data <- cpi_core_data |>
  mutate(ms_weight = ms_ri / core_ri)

# --------------------------------------------------------------------------
# Step 4: Contribution = weight * component inflation
# --------------------------------------------------------------------------
cpi_core_data <- cpi_core_data |>
  mutate(ms_contrib = ms_weight * ms_infl)

# --------------------------------------------------------------------------
# Step 5: Medical Care Services' share of Core CPI inflation
# --------------------------------------------------------------------------
cpi_core_data <- cpi_core_data |>
  mutate(ms_contrib_share = (ms_contrib / core_infl) * 100)

# --------------------------------------------------------------------------
# Pull each series and extract the last non-NA date
# --------------------------------------------------------------------------
get_end_date <- function(code, db) {
  full_code <- paste0(code, "@", db)
  tryCatch({
    d <- import_haver(series = full_code, eop = FALSE) |>
      na.omit()
    col <- setdiff(names(d), "date")
    d <- d |> filter(!is.na(.data[[col[1]]]))
    if (nrow(d) == 0) return(as.Date(NA))
    max(d$date)
  }, error = function(e) as.Date(NA))
}

staleness <- series_registry |>
  rowwise() |>
  mutate(end_date = get_end_date(series_code, database)) |>
  ungroup()

# --------------------------------------------------------------------------
# Determine the most recent end date and flag stale series
# --------------------------------------------------------------------------
most_recent <- max(staleness$end_date, na.rm = TRUE)

staleness <- staleness |>
  mutate(
    months_behind = as.numeric(
      (year(most_recent) - year(end_date)) * 12 +
      (month(most_recent) - month(end_date))
    ),
    status = case_when(
      is.na(end_date)       ~ "Stale",
      months_behind == 0    ~ "Current",
      months_behind == 1    ~ "Behind",
      months_behind <= 5    ~ "Delayed",
      TRUE                  ~ "Stale"
    )
  )

# --------------------------------------------------------------------------
# Single color-coded table for all series
# --------------------------------------------------------------------------
staleness_display <- staleness |>
  select(source_group, series_code, database, description, end_date, status) |>
  arrange(status, source_group, series_code)

DT::datatable(
  staleness_display,
  caption = paste0(
    "Series Staleness Monitor (most recent: ",
    format(most_recent, "%B %Y"), ")"
  ),
  colnames = c("Source", "Code", "Database", "Description", "End Date", "Status"),
  rownames = FALSE,
  extensions = 'Buttons',
  options = list(
    scrollY = T,
    dom = 'Blfrtip',
    buttons = c('copy', 'csv', 'excel'),
    pageLength = 10,
    lengthMenu = list(c(10, 25, 50, -1), c('10', '25', '50', 'All'))
  )
) |>
  DT::formatStyle(
    "status",
    target = "row",
    backgroundColor = DT::styleEqual(
      c("Current", "Behind", "Delayed", "Stale"),
      c("#d4edda", "#fff3cd", "#ffd6a5", "#f8d7da")
    )
  )
